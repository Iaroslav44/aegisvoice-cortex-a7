cmake_minimum_required(VERSION 3.20)
project(AegisVoice VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(AV_SANITIZE "Enable address and undefined behaviour sanitizers" OFF)
option(AV_BUILD_TESTS "Build host verification" ON)
option(AV_BUILD_HOST "Build host command-line audio demonstration" ON)

add_library(av_warnings INTERFACE)
if(MSVC)
    target_compile_options(av_warnings INTERFACE /W4 /WX)
else()
    target_compile_options(av_warnings INTERFACE -Wall -Wextra -Wpedantic -Werror
        -Wconversion -Wshadow -Wstrict-prototypes)
endif()

if(AV_SANITIZE)
    if(MSVC)
        message(FATAL_ERROR "Use GCC or Clang for AV_SANITIZE (ASan + UBSan)")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()

add_library(monocypher STATIC
    third_party/monocypher/monocypher.c
    third_party/monocypher/monocypher-ed25519.c)
target_include_directories(monocypher PUBLIC third_party/monocypher)

add_library(aegisvoice STATIC src/audio.c src/transport.c src/secure.c src/device.c)
target_include_directories(aegisvoice PUBLIC include)
target_link_libraries(aegisvoice PUBLIC monocypher av_warnings)

add_library(av_firmware_adapter STATIC firmware/av_core_adapter.c)
target_include_directories(av_firmware_adapter PUBLIC firmware)
target_link_libraries(av_firmware_adapter PUBLIC aegisvoice)

if(AV_BUILD_HOST)
    add_executable(aegisvoice_demo host/demo.c host/random.c host/wav.c)
    target_include_directories(aegisvoice_demo PRIVATE host)
    target_link_libraries(aegisvoice_demo PRIVATE aegisvoice)
    if(WIN32)
        target_link_libraries(aegisvoice_demo PRIVATE bcrypt)
    endif()
endif()

if(AV_BUILD_TESTS)
    enable_testing()
    foreach(name audio transport secure)
        add_executable(test_${name} tests/test_${name}.c)
        target_link_libraries(test_${name} PRIVATE aegisvoice)
        add_test(NAME ${name} COMMAND test_${name})
    endforeach()
    add_executable(test_device tests/test_device.c)
    target_link_libraries(test_device PRIVATE av_firmware_adapter)
    add_test(NAME device_adapter COMMAND test_device)
    add_executable(test_firmware_runtime tests/test_firmware_runtime.c)
    target_include_directories(test_firmware_runtime PRIVATE firmware/check include)
    target_link_libraries(test_firmware_runtime PRIVATE av_warnings)
    add_test(NAME firmware_runtime COMMAND test_firmware_runtime)
    if(AV_BUILD_HOST)
        add_test(NAME end_to_end COMMAND aegisvoice_demo --self-test)
        find_package(Python3 COMPONENTS Interpreter QUIET)
        if(Python3_Interpreter_FOUND)
            add_test(NAME host_wav_cli COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_host.py $<TARGET_FILE:aegisvoice_demo>)
        endif()
    endif()
endif()
